function localAnalyze(){
 const p=panel(),item=itemById(rec.itemId);if(!p||!item)return;
 const text=(rec.finalText+rec.interim).trim();const words=(text.match(/[A-Za-z]+(?:'[A-Za-z]+)?/g)||[]);
 const wpm=rec.duration?Math.round(words.length/(rec.duration/60)):0;
 const fillers=(text.toLowerCase().match(/\b(um+|uh+|erm+|you know|i mean)\b/g)||[]).length;
 const repeat=(text.toLowerCase().match(/\b([a-z]+)\s+\1\b/g)||[]).length;
 const keyText=(item.part==='p1'?item.keys:item.keys.map(k=>k.text)).join(' ').toLowerCase();
 const stops=new Set('the a an and or but to of in on at for with is are was were i my me it this that do does did have has had be been being'.split(' '));
 const keyWords=[...new Set((keyText.match(/[a-z]+/g)||[]).filter(w=>w.length>3&&!stops.has(w)))];
 const hit=keyWords.filter(w=>text.toLowerCase().includes(w));const coverage=keyWords.length?Math.round(hit.length/keyWords.length*100):0;
 const pace=!text?'没有转写，无法计算':wpm<90?'偏慢，注意不要长时间搜索内容':wpm>170?'偏快，注意清晰度和停顿':'速度较稳定';
 const box=p.querySelector('.local-analysis');box.hidden=false;
 box.innerHTML=`<b>即时练习数据（不是雅思评分）</b><ul><li>时长：${fmt(rec.duration)}；转写词数：${words.length}；约 ${wpm||'—'} 词/分钟（${pace}）</li><li>超过约1.5秒的静音段：${rec.pauses}；明显填充词：${fillers}；连续重复词：${repeat}</li><li>与当前答案触发词的字面覆盖：${text?coverage+'%':'—'}。改用自己的表达时，这项可能低估。</li></ul>${text?`<details><summary>查看转写</summary><p>${esc(text)}</p></details>`:'<p>当前浏览器没有提供实时转写。仍可下载录音后上传到聊天中分析。</p>'}`;
}
