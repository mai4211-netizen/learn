function p1Memory(title){
 const q=(title||'').toLowerCase().replace(/[?]/g,'').trim();
 if(/where is your hometown|where is your home|where do you live/.test(q))return'地点 → 方位或周边 → 最明显的特点';
 if(/what is your hometown like|what.*place.*like|what.*home.*like/.test(q))return'整体印象 → 两个具体特点 → 一个代表性细节';
 if(/^what do you do|work or study/.test(q))return'身份 → 日常工作内容 → 最终做出的东西';
 if(/what.*interesting|most interesting/.test(q))return'有趣点 → 为什么有趣 → 一个具体例子';
 if(/why/.test(q))return'直接原因 → 对我的影响 → 一个日常细节';
 if(/how often|how much time|spend a lot of time/.test(q))return'频率 → 常见场景 → 为什么是这个频率';
 if(/when did|when do|what time/.test(q))return'时间 → 当时做什么 → 为什么选这个时间';
 if(/where do|where did/.test(q))return'地点 → 当时做什么 → 为什么在那里';
 if(/have you ever|did you|were you|was there/.test(q))return'先回答有没有 → 一次具体经历 → 感受或结果';
 if(/do you prefer|which do you prefer|would you rather/.test(q))return'先选一边 → 解释取舍 → 和另一边对比';
 if(/do you like|are you interested|enjoy|favourite|favorite/.test(q))return'先表态 → 最主要原因 → 一个具体场景';
 if(/do you think|should|could.*improv|how could|how can/.test(q))return'先给判断 → 问题或原因 → 一个具体做法';
 if(/what kind|what type|which kind/.test(q))return'类型 → 为什么喜欢 → 一个具体例子';
 if(/^what/.test(q))return'直接回答 → 一个特点 → 一个具体细节';
 if(/^how/.test(q))return'简短结论 → 过程或程度 → 一个例子';
 return'直接回答 → 解释原因 → 一个具体细节';
}
